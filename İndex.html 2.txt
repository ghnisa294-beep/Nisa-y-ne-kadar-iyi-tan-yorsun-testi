let current=0, score=0, answered=false;

function startGame(){
 current=0; score=0;
 document.getElementById("startScreen").classList.add("hidden");
 document.getElementById("resultScreen").classList.add("hidden");
 document.getElementById("quizScreen").classList.remove("hidden");
 showQuestion();
}

function showQuestion(){
 answered=false;
 const q=questions[current];
 document.getElementById("questionNo").textContent=`SORU ${current+1} / 20`;
 document.getElementById("score").textContent=`SKOR: ${score}`;
 document.getElementById("fill").style.width=`${((current+1)/20)*100}%`;
 document.getElementById("question").textContent=q[0];
 document.getElementById("feedback").textContent="";
 const box=document.getElementById("choices");
 box.innerHTML="";
 ["A","B","C","D"].forEach((letter,i)=>{
   const b=document.createElement("button");
   b.className="choice";
   b.textContent=`${letter}) ${q[1][i]}`;
   b.onclick=()=>answer(i,b);
   box.appendChild(b);
 });
}

function answer(i,button){
 if(answered)return;
 answered=true;
 const q=questions[current];
 const buttons=document.querySelectorAll(".choice");
 buttons[q[2]].classList.add("correct");
 if(i===q[2]){
   score++;
   button.classList.add("correct");
   document.getElementById("feedback").textContent="DOĞRU! 🎉";
   document.getElementById("feedback").style.color="#45a86b";
 }else{
   button.classList.add("wrong");
   document.getElementById("feedback").textContent="YANLIŞ! 😭";
   document.getElementById("feedback").style.color="#e45d5d";
 }
 document.getElementById("score").textContent=`SKOR: ${score}`;
 setTimeout(()=>{
   current++;
   if(current<questions.length) showQuestion();
   else showResult();
 },900);
}

function showResult(){
 document.getElementById("quizScreen").classList.add("hidden");
 document.getElementById("resultScreen").classList.remove("hidden");
 const percent=Math.round(score/20*100);
 document.getElementById("finalScore").textContent=`${score} / 20`;
 document.getElementById("percent").textContent=`Başarı: %${percent}`;
 let text="";
 if(score===20) text="🏆 EFSANE! Nisa'yı senden iyi tanıyan yok!";
 else if(score>=17) text="🥇 MÜKEMMEL! Nisa hakkında bayağı bilgi sahibisin.";
 else if(score>=14) text="🥈 ÇOK İYİ! Nisa seni sever.";
 else if(score>=10) text="🥉 ORTA! Biraz daha Nisa bilgisi lazım.";
 else if(score>=5) text="😅 ZAYIF! Nisa'yı biraz daha tanıman lazım.";
 else text="💀 FELAKET! Bu testte nasıl bu kadar dağıldın?";
 document.getElementById("resultText").textContent=text;
}
